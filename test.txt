	//                   และจะมีตัวแปรชื่อ test แทน instance ของ Info ที่เรียกเมธอดนี้
	// something       → ชื่อของเมธอดนี้

	// POST /send
func SendEncrypted(c *gin.Context) {
    message := "this is a secret"
    encrypted, _ := rsa.EncryptOAEP(sha256.New(), rand.Reader, &publicKey, []byte(message), nil)
    c.JSON(200, gin.H{"data": base64.StdEncoding.EncodeToString(encrypted)})
}

// POST /receive
func ReceiveEncrypted(c *gin.Context) {
    var json struct {
        Data string `json:"data"`
    }
    c.BindJSON(&json)

    encryptedBytes, _ := base64.StdEncoding.DecodeString(json.Data)
    decrypted, _ := privateKey.Decrypt(nil, encryptedBytes, &rsa.OAEPOptions{Hash: crypto.SHA256})
    c.JSON(200, gin.H{"message": string(decrypted)})
}
